# Refonte meriembr.com : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remplacer l'ancien template Bootstrap par le nouveau portfolio ludique illustré de Meriem Ben Rejeb, en FR et EN, statique, servi par GitHub Pages.

**Architecture:** Deux pages HTML statiques (`index.html`, `en/index.html`) sans framework. Un script Python (`outils/generer.py`) lit `data/travaux.json` et `data/clients.json`, crée les miniatures WebP et injecte le HTML de la grille, des filtres et des logos clients dans les deux pages entre des marqueurs `<!-- x:debut -->` / `<!-- x:fin -->`. Le JS (`js/site.js`, module ES) ne fait qu'ajouter filtres, visionneuse et animations sur un HTML déjà complet : sans JS, tout reste lisible.

**Tech Stack:** HTML5, CSS (variables, grid, `clamp()`), JS ES modules natifs, Python 3 + Pillow (déjà installé, WebP animé disponible), Node 22 (`node --test`) pour les fonctions JS pures, Playwright (via `uv run`) pour la vérification.

**Spec:** `docs/specs/2026-11-04-refonte-design.md` (lire en entier avant toute tâche).

```python
"""Génère les miniatures WebP et le HTML injecté dans index.html et en/index.html.

Lancer depuis la racine du dépôt : python3 -I outils/generer.py
Source éditable : data/travaux.json (client, grand) et data/clients.json.
"""
import html
import json
import re
import sys
from pathlib import Path

from PIL import Image

CATEGORIES = [
    ("logos", "Logos", "Logos"),
    ("visual_identity", "Identité visuelle", "Visual identity"),
    ("flyers_leaflets", "Flyers et dépliants", "Flyers and leaflets"),
    ("books_magazines", "Livres et magazines", "Books and magazines"),
    ("packaging_labels", "Packaging et étiquettes", "Packaging and labels"),
    ("various", "Divers", "Various"),
    ("web", "Web", "Web"),
    ("pos_panels_dressing", "PLV et habillage", "POS and dressing"),
    ("gaming", "Gaming", "Gaming"),
]
TAILLE_MIN = 600


def numero(nom):
    m = re.match(r"(\d+)", nom)
    return int(m.group(1)) if m else 0


def deviner_client(nom):
    reste = re.sub(r"^\d+-?", "", nom)
    return " ".join(m.capitalize() for m in re.split(r"[-_]+", reste) if m)


def miniature(src, dst):
    dst.parent.mkdir(parents=True, exist_ok=True)
    im = Image.open(src)
    if getattr(im, "n_frames", 1) > 1:
        images = []
        for i in range(im.n_frames):
            im.seek(i)
            f = im.convert("RGBA")
            f.thumbnail((TAILLE_MIN, TAILLE_MIN))
            images.append(f)
        images[0].save(dst, "WEBP", save_all=True, append_images=images[1:],
                       duration=im.info.get("duration", 100), loop=0, quality=80)
    else:
        im = im.convert("RGB")
        im.thumbnail((TAILLE_MIN, TAILLE_MIN))
        im.save(dst, "WEBP", quality=80)


def charger_travaux(racine):
    existant = {}
    fj = racine / "data/travaux.json"
    if fj.exists():
        existant = {t["fichier"]: t for t in json.loads(fj.read_text(encoding="utf-8"))}
    travaux = []
    for cle, _, _ in CATEGORIES:
        dossier = racine / "img/portfolio" / cle
        fichiers = [p for p in dossier.glob("*") if p.suffix in (".jpg", ".gif")] if dossier.exists() else []
        for i, p in enumerate(sorted(fichiers, key=lambda p: (numero(p.stem), p.name))):
            rel = p.relative_to(racine).as_posix()
            pdf = f"pdf/{cle}/{p.stem}.pdf"
            ancien = existant.get(rel, {})
            travaux.append({
                "fichier": rel,
                "min": f"img/min/{cle}/{p.stem}.webp",
                "categorie": cle,
                "client": ancien.get("client", deviner_client(p.stem)),
                "grand": ancien.get("grand", i == 0),
                "pdf": pdf if (racine / pdf).exists() else None,
            })
    return travaux


def injecter(texte, nom, contenu):
    debut, fin = f"<!-- {nom}:debut -->", f"<!-- {nom}:fin -->"
    if debut not in texte or fin not in texte:
        raise ValueError(f"marqueur {nom} absent")
    a = texte.index(debut) + len(debut)
    b = texte.index(fin)
    return texte[:a] + contenu + texte[b:]
```

```python
"""Liste les liens locaux cassés (src, href, url()) des pages et CSS donnés.
Lancer : python3 -I outils/verifier.py index.html en/index.html css/site.css
Code de sortie 1 si un lien est cassé."""
import re
import sys
import urllib.parse
from pathlib import Path

casses = []
for f in map(Path, sys.argv[1:]):
    texte = f.read_text(encoding="utf-8")
    for m in re.finditer(r'(?:src|href)="([^"]+)"|url\(["\']?([^"\')]+)', texte):
        u = m.group(1) or m.group(2)
        if re.match(r"(https?:|mailto:|tel:|data:|#)", u):
            continue
        chemin = urllib.parse.unquote(u.split("#")[0].split("?")[0])
        if chemin and not (f.parent / chemin).exists():
            casses.append(f"{f}: {u}")
print("\n".join(casses) or "0 lien cassé")
sys.exit(1 if casses else 0)
```

```js
export function lireHash(hash) {
  const m = /^#travaux\/([a-z_]+)$/.exec(hash);
  return m ? m[1] : "tout";
}

export function visibles(items, cat) {
  return items.flatMap((it, i) => (cat === "tout" || it.cat === cat ? [i] : []));
}

export function voisin(position, total, pas) {
  return (position + pas + total) % total;
}
```

```python
def main(racine=Path(".")):
    travaux = charger_travaux(racine)
    for t in travaux:
        src, dst = racine / t["fichier"], racine / t["min"]
        if not dst.exists() or dst.stat().st_mtime < src.stat().st_mtime:

    for page, langue, prefixe in PAGES:

        f.write_text(h, encoding="utf-8")
    print(len(travaux), "travaux,", len(clients), "clients")
```

```js
import { lireHash, visibles, voisin } from "./galerie.js";

const grille = document.getElementById("grille");
const vignettes = [...grille.querySelectorAll(".vignette")];
const items = vignettes.map((v) => ({ cat: v.dataset.cat }));
const boutons = [...document.querySelectorAll("#filtres [data-cat]")];
let courant = "tout";

function filtrer(cat) {
  courant = boutons.some((b) => b.dataset.cat === cat) ? cat : "tout";
  const ok = new Set(visibles(items, courant));
  vignettes.forEach((v, i) => { v.hidden = !ok.has(i); });
  boutons.forEach((b) => b.setAttribute("aria-pressed", String(b.dataset.cat === courant)));
}

boutons.forEach((b) => b.addEventListener("click", () => {
  const cat = b.dataset.cat;
  history.pushState(null, "", cat === "tout" ? "#travaux" : `#travaux/${cat}`);
  filtrer(cat);
}));
addEventListener("popstate", () => filtrer(lireHash(location.hash)));
filtrer(lireHash(location.hash));
if (location.hash.startsWith("#travaux/")) document.getElementById("travaux").scrollIntoView();

/* Visionneuse */
const vis = document.getElementById("visionneuse");
const visImg = vis.querySelector(".vis-image");
const visLegende = vis.querySelector(".vis-legende");
const visCompteur = vis.querySelector(".vis-compteur");
const visPdf = vis.querySelector(".vis-pdf");
let liste = [];
let pos = 0;
let origine = null;

function afficher() {
  const lien = vignettes[liste[pos]].querySelector("a");
  const alt = lien.querySelector("img").alt;
  visImg.src = lien.href;
  visImg.alt = alt;
  visLegende.textContent = alt;
  visCompteur.textContent = `${pos + 1} / ${liste.length}`;
  visPdf.hidden = !lien.dataset.pdf;
  if (lien.dataset.pdf) visPdf.href = lien.dataset.pdf;
}

grille.addEventListener("click", (e) => {
  const lien = e.target.closest(".vignette a");
  if (!lien || e.ctrlKey || e.metaKey || e.shiftKey) return;
  e.preventDefault();
  liste = visibles(items, courant);
  pos = liste.indexOf(vignettes.indexOf(lien.parentElement));
  origine = lien;
  afficher();
  vis.showModal();
});
const pas = (n) => { pos = voisin(pos, liste.length, n); afficher(); };
vis.querySelector(".vis-prec").addEventListener("click", () => pas(-1));
vis.querySelector(".vis-suiv").addEventListener("click", () => pas(1));
vis.querySelector(".vis-fermer").addEventListener("click", () => vis.close());
vis.addEventListener("click", (e) => { if (e.target === vis) vis.close(); });
vis.addEventListener("keydown", (e) => {
  if (e.key === "ArrowLeft") pas(-1);
  if (e.key === "ArrowRight") pas(1);
});
vis.addEventListener("close", () => origine?.focus());
let x0 = null;
vis.addEventListener("touchstart", (e) => { x0 = e.touches[0].clientX; }, { passive: true });
vis.addEventListener("touchend", (e) => {
  if (x0 === null) return;
  const dx = e.changedTouches[0].clientX - x0;
  if (Math.abs(dx) > 50) pas(dx < 0 ? 1 : -1);
  x0 = null;
});

/* Menu mobile */
const menuBouton = document.getElementById("menu-bouton");
const menu = document.getElementById("menu");
menuBouton.addEventListener("click", () => {
  const ouvert = menu.classList.toggle("ouvert");
  menuBouton.setAttribute("aria-expanded", String(ouvert));
});
menu.addEventListener("click", (e) => {
  if (e.target.closest("a")) { menu.classList.remove("ouvert"); menuBouton.setAttribute("aria-expanded", "false"); }
});

/* Apparitions et tracé du chemin */
const calme = matchMedia("(prefers-reduced-motion: reduce)").matches;
const trait = document.getElementById("chemin-trait");
if (!calme && trait) {
  const long = trait.getTotalLength();
  trait.style.strokeDasharray = long;
  trait.style.strokeDashoffset = long;
  const dessiner = () => {
    const r = trait.closest(".chemin").getBoundingClientRect();
    const avance = Math.min(1, Math.max(0, (innerHeight * 0.8 - r.top) / r.height));
    trait.style.strokeDashoffset = long * (1 - avance);
  };
  addEventListener("scroll", dessiner, { passive: true });
  dessiner();
}
const obs = new IntersectionObserver((entrees) => entrees.forEach((e) => {
  if (e.isIntersecting) { e.target.classList.add("vu"); obs.unobserve(e.target); }
}), { threshold: 0.15 });
document.querySelectorAll(".visible-au-scroll").forEach((el) => obs.observe(el));
```
